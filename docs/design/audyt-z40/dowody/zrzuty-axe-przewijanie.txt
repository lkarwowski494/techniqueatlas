const { chromium } = require('playwright');
const fs = require('fs'), path = require('path');
const [out, w, strony, motywy, kontrast, skala = '2', axeOn = '1', maxH = '6000'] = process.argv.slice(2);

const lista = fs.readFileSync(strony.slice(1), 'utf8').trim().split('\n').map(s => s.split('='));
(async () => {
  fs.mkdirSync(out, { recursive: true });
  const b = await chromium.launch(); const wynik = [];

    const ctx = await b.newContext({ viewport: { width, height: 900 }, deviceScaleFactor: Number(skala), colorScheme: theme, reducedMotion: 'reduce', contrast: kontrast });

    for (const [name, url] of lista) {
      await p.goto('http://localhost:8124' + url); await p.evaluate(() => document.fonts.ready);
      const poziom = await p.evaluate(() => document.documentElement.scrollWidth - document.documentElement.clientWidth);

      }
      const tag = `${name}-${width}-${theme === 'light' ? 'jasny' : 'ciemny'}${kontrast === 'more' ? '-kontrast' : ''}`;
      const h = await p.evaluate(() => document.documentElement.scrollHeight);
      const opt = { path: path.join(out, `${tag}.png`), fullPage: true };
